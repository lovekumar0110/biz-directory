app.get('/api/items',(q,r)=>{
 const{s='',cat='',p=1,n=12}=q.query;
 const d=rd().filter(i=>(i.name+i.desc).toLowerCase().includes(s.toLowerCase())&&(!cat||i.cat===cat));

});

});
